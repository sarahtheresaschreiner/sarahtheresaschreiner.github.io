    Wifi mit esp 32

    Command Beschreibung Parameter
    #include <WiFi.h> library
    WiFi.mode(mode)
    WiFi.begin(ssid, password) mit wlan connecten ssid = name
    WiFi.status() != WL_CONNECTED schauen ob die Verbindung schon aufgebaut ist
    WiFi.RSSI() Wifi strengh
    WiFi.localIP() returned die IP im Wifi
    WiFi.config(local_IP, gateway, subnet) static ip
    WiFi.disconnect()
    WiFi.reconnect() nach dem die Verbindung verloren geht
    WiFi.softAP(ssid, password)
    WiFi.softAPdisconnect()
    WiFi.softAPIP() return IP of this

    Modes

    • WIFI_STA: esp connects to an access point
    • WIFI_AP: Access Point: stations can connect to the esp
    • WIFI_AP_STA: both

    Status

    • WL_CONNECT_FAILED

    Check wether die WLAN-Verbindung noch steht

    unsigned long currentMillis = millis(); 
// if WiFi is down, try reconnecting 
if ((WiFi.status() != WL_CONNECTED) && (currentMillis - previousMillis >=interval)) {
     Serial.print(millis()); 
     Serial.println("Reconnecting to WiFi...");
     WiFi.disconnect(); 
     WiFi.reconnect(); 
     previousMillis = currentMillis; 
}

    Client (TCP)

    Befehl Funktion
    WiFiClient client; TCP-Client erstellen
    client.connect(ip, port) Verbindung zum Server herstellen
    client.connected() Prüfen, ob Verbindung besteht
    client.write(data) Daten als Bytes senden
    client.print(data) Text senden
    client.println(data) Text mit Zeilenumbruch senden
    client.available() Anzahl lesbarer Bytes
    client.read() Ein Byte lesen
    client.readString() Empfangene Daten als String lesen
    client.stop() Verbindung schließen

    WebSocket-Client-Befehle

    Für deine geplante Kommunikation mit Javalin empfehle ich WebSocketsClient aus der Bibliothek arduinoWebSockets von Links2004.

    Befehl Funktion
    WebSocketsClient webSocket; WebSocket-Client erstellen
    webSocket.begin(ip, port, path) Verbindung zum Server aufbauen
    webSocket.onEvent(callback) Event-Handler registrieren
    webSocket.loop() Verbindung und Nachrichten verarbeiten
    webSocket.sendTXT(data) Textnachricht senden
    webSocket.sendBIN(data, len) Binärdaten senden
    webSocket.disconnect() Verbindung trennen
    webSocket.isConnected() Verbindungsstatus prüfen

    Wichtige Events:

    Event Bedeutung
    WStype_CONNECTED Verbindung aufgebaut
    WStype_DISCONNECTED Verbindung verloren
    WStype_TEXT Textnachricht empfangen
    WStype_BIN Binärnachricht empfangen
    WStype_ERROR Fehler aufgetreten

    Beispiel der Verbindung

    Hier ein minimales Beispiel für deinen ESP32 als Access Point und WebSocket-Client.

    #include <WiFi.h>
#include <WebSocketsClient.h>

// WLAN
const char* ssid = "MeinAuto";
const char* password = "12345678";

// Javalin-Server
const char* serverIP = "192.168.4.2";
const int serverPort = 8080;

WebSocketsClient webSocket;

// WebSocket Events
void webSocketEvent(WStype_t type, uint8_t* payload, size_t length) {

    switch (type) {

        case WStype_CONNECTED:
            Serial.println("Javalin verbunden!");
            break;

        case WStype_DISCONNECTED:
            Serial.println("Verbindung getrennt!");
            break;

        case WStype_TEXT:
            Serial.printf("Empfangen: %s\n", payload);
            break;
    }
}

void setup() {
    Serial.begin(115200);

    // Access Point erstellen
    WiFi.mode(WIFI_AP);
    WiFi.softAP(ssid, password);

    Serial.println(WiFi.softAPIP());

    // WebSocket verbinden
    webSocket.begin(serverIP, serverPort, "/car");
    webSocket.onEvent(webSocketEvent);
}

void loop() {
    // WebSocket am Leben halten und Events verarbeiten
    webSocket.loop();
}
